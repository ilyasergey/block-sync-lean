Section 4.2 protocol

-- Step 1: Validator proposes a block (broadcasts it)
| proposeBlock (state : DAGCertifiedState) (validatorId : ValidatorId) (block : Block)
               (newState : DAGCertifiedState) :
  -- Preconditions
  block.author = validatorId →
  block.r = state.currentRound →
  -- Block references at least 2f+1 parents from previous round
  block.parents.length ≥ 2 * system.f + 1 →
  -- New state: block added to pending blocks
  newState = { state with
    pendingBlocks := block :: state.pendingBlocks
    emittedOperations := (.block_propose validatorId block block.r) :: state.emittedOperations } →
  DAGCertifiedTransition state (.basicOp (.block_propose validatorId block block.r)) newState

-- Step 2: Validator signs a block (responds with signature)
| signBlock (state : DAGCertifiedState) (validatorId : ValidatorId) (blockDigest : BlockDigest)
  (newState : DAGCertifiedState)
  existingSigners newSigners (updatedCerts : List (BlockDigest × List ValidatorId)) :
  -- Preconditions
  isPendingBlock state blockDigest →
  -- Validator hasn't already signed this block
  (∀ cert ∈ state.certificates, blockDigest ≠ cert.1 ∨ validatorId ∉ cert.2) →
  -- Explicit parameter equalities
  existingSigners = (getCertificateSigners state blockDigest |>.getD []) →
  newSigners = validatorId :: existingSigners →
  updatedCerts = state.certificates.filter (fun (d, _) => d ≠ blockDigest) →
  -- New state: add signature to certificate
  newState = { state with
    certificates := (blockDigest, newSigners) :: updatedCerts
    dagOperations := (.signBlock validatorId blockDigest) :: state.dagOperations } →
  DAGCertifiedTransition state (.dagOp (.signBlock validatorId blockDigest)) newState

-- Step 3: Validator broadcasts certificate and stores block
| storeBlock (state : DAGCertifiedState) (validatorId : ValidatorId) (block : Block)
  (newState : DAGCertifiedState)
  (newBlocks : List Block) (newPendingBlocks : List Block) (shouldAdvanceRound : Bool)
  (signers : List ValidatorId) :
  -- Preconditions
  hasCertificate system state block.d →
  isPendingBlock state block.d →
  -- Block not already in accepted blocks
  ¬(block ∈ state.blocks) →
  -- Explicit parameter equalities
  newBlocks = block :: state.blocks →
  newPendingBlocks = state.pendingBlocks.filter (fun b => b.d ≠ block.d) →
  shouldAdvanceRound = ((newBlocks.filter (fun b => b.r = state.currentRound)).length ≥ 2 * system.f + 1) →
  signers = (getCertificateSigners state block.d |>.getD []) →
  -- New state: move block from pending to accepted, advance round if conditions met
  newState = { state with
    blocks := newBlocks
    pendingBlocks := newPendingBlocks
    currentRound := if shouldAdvanceRound then state.currentRound + 1 else state.currentRound
    emittedOperations := (ValidatorOperation.block_store validatorId block) :: state.emittedOperations
    dagOperations := (.broadcastCertificate validatorId block.d signers) :: state.dagOperations } →
  DAGCertifiedTransition state (.basicOp (ValidatorOperation.block_store validatorId block)) newState

-- Deterministic pull: Request missing block
| requestBlock (state : DAGCertifiedState) (requesterId : ValidatorId) (blockDigest : BlockDigest)
  (newState : DAGCertifiedState) :
  -- Preconditions
  -- Block is referenced but not available locally
  (∃ b ∈ state.blocks ++ state.pendingBlocks, blockDigest ∈ b.parents) →
  ¬(∃ b ∈ state.blocks ++ state.pendingBlocks, b.d = blockDigest) →
  -- Request not already pending
  ¬((requesterId, blockDigest) ∈ state.pendingRequests) →
  -- New state: add to pending requests and DAG operations
  newState = { state with
    pendingRequests := (requesterId, blockDigest) :: state.pendingRequests
    dagOperations := (DAGCertifiedOperation.requestMissingBlock requesterId blockDigest) :: state.dagOperations } →
  DAGCertifiedTransition state (DAGCertifiedTransitionOp.dagOp (DAGCertifiedOperation.requestMissingBlock requesterId blockDigest)) newState

-- Respond to pull request: Provide block
| provideBlock (state : DAGCertifiedState) (providerId : ValidatorId) (block : Block)
  (requesterId : ValidatorId) (newState : DAGCertifiedState) :
  -- Preconditions
  -- Provider has the block and there was a request
  block ∈ state.blocks →
  (requesterId, block.d) ∈ state.pendingRequests →
  -- New state: add to pending provisions and DAG operations
  newState = { state with
    pendingProvisions := (providerId, requesterId, block) :: state.pendingProvisions
    dagOperations := (DAGCertifiedOperation.provideMissingBlock providerId requesterId block) :: state.dagOperations } →
  DAGCertifiedTransition state (DAGCertifiedTransitionOp.dagOp (DAGCertifiedOperation.provideMissingBlock providerId requesterId block)) newState

-- Receive block from pull: Store provided block
| receiveBlock (state : DAGCertifiedState) (receiverId : ValidatorId) (block : Block)
  (newState : DAGCertifiedState)
  (newState' : DAGCertifiedState) (finalState : DAGCertifiedState) :
  -- Preconditions
  -- There was a provide operation for this block to this receiver
  (∃ providerId, (providerId, receiverId, block) ∈ state.pendingProvisions) →
  -- Block not already stored
  ¬(block ∈ state.blocks ++ state.pendingBlocks) →
  -- Explicit parameter equalities
  newState' = (if hasCertificate system state block.d
    then { state with blocks := block :: state.blocks }
    else { state with pendingBlocks := block :: state.pendingBlocks }) →
  finalState = { newState' with
    pendingRequests := state.pendingRequests.filter (fun (rid, bd) => ¬(rid = receiverId ∧ bd = block.d))
    pendingProvisions := state.pendingProvisions.filter (fun (_, rid, b) => ¬(rid = receiverId ∧ b.d = block.d)) } →
  newState = finalState →
  DAGCertifiedTransition state (DAGCertifiedTransitionOp.basicOp (ValidatorOperation.block_store receiverId block)) newState
